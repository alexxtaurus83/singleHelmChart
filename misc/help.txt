{{ $actualCertName := "blank"}}
  {{- $appCertName := printf "venafi-%s-%s" $.Values.applicationName ( . | lower ) -}}
  {{- /* Check for secret with name maching app-specific cert name.  Assumption is this means cert object is in ready state and can be mounted. */}}
  {{- $existingAppCert := (lookup "cert-manager.io/v1" "Certificate" $.Release.Namespace $appCertName) }}
  {{- /* Keep app specific spid certifiicate object if one exists */}}
  {{- if $existingAppCert }}
    {{ $actualCertName = $appCertName}}
  {{- else }}
    {{- /*
        Check if a generic spid certificate exists.
        If a generic spid certificate does not exist or is owned by this "helm release" include the generic spid certificate object definition in output
        objects for this chart so it is created/remains deployed.  Note that the actual secret object generated by the certificate object is not directly
        controlled by the helm chart, therefore uninstalling the helm release that officially owns the certificate will not delete the secret and break
        the other deployed apps.  We should do our best to kee the certificate object present in the namespace so the certificate operater can properly
        manage/renew the cert & secret in the future.
    */}}
    {{- $genericCertName := printf "venafi-%s" ( . | lower ) -}}
    {{- $existingGenericCert := (lookup "cert-manager.io/v1" "Certificate" $.Release.Namespace $genericCertName) | default dict}}
    {{- $genericCertMetadata := (get $existingGenericCert "metadata") | default dict}}
    {{- $genericCertAnnotations := (get $genericCertMetadata "annotations") | default dict}}
    {{- $genericCertReleaseName := (get $genericCertAnnotations "meta.helm.sh/release-name") | default "none"}}
    {{- if or (not $existingGenericCert) (eq $genericCertReleaseName $.Release.Name) }}
      {{ $actualCertName = $genericCertName }}
    {{- end }}
  {{- end }}
